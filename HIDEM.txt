# Base para todos os modelos: Friedmann Equation E(z)
# E(z) = H(z) / H0
def E_squared(z, Omega_m, Omega_de_term):
    """
    Retorna E(z)^2 = (H(z)/H0)^2 para um universo plano.
    Omega_de_term é a função que retorna a contribuição da energia escura.
    """
    return Omega_m * (1 + z)**3 + Omega_de_term(z, Omega_m) + (1 - Omega_m - Omega_de_term(0, Omega_m)) # (1 - Omega_m - Omega_de0) ajusta para curvatura se Omega_de_term(0) é Omega_de0

# 2.1. Lambda-CDM Model
def Omega_de_LCDM(z, Omega_m):
    return 1 - Omega_m # Asumindo Omega_Lambda = 1 - Omega_m para universo plano

# 2.2. CPL (Chevallier-Polarski-Linder) Model
def Omega_de_CPL(z, Omega_m, w0, wa):
    # w(z) = w0 + wa * z / (1 + z)
    # Omega_DE(z) = Omega_DE0 * exp( -3 * integral_0^z (1+w(x))/(1+x) dx )
    Omega_de0 = 1 - Omega_m # Assuming Omega_de0 is the remaining part at z=0 for flatness
    integrand = lambda zp: (1 + (w0 + wa * zp / (1 + zp))) / (1 + zp)
    integral, _ = quad(integrand, 0, z)
    return Omega_de0 * np.exp(-3 * integral)

# 2.3. EDE (Early Dark Energy) Model (similar ao de hybrid_cosmology.py)
def Omega_de_EDE(z, Omega_m, f_EDE, z_c):
    # Este é o termo de EDE do hybrid_cosmology, que parece combinar com CPL
    # Para simplificar, vou considerar EDE puro se w0 = -1, wa = 0 no CPL
    # ou uma versão simplificada de EDE.
    # A forma em hybrid_cosmology.py é mais complexa e combina CPL com tanh.
    # Vamos usar uma versão mais comum para EDE, onde Omega_DE(z) varia.
    # Se f_EDE é a fração da energia escura total no início
    # e z_c o redshift de transição.
    # Usando a forma de 'friedmann_model_fit.py.txt':
    # Omega_EDE(z, Omega_EDE0, zc, m=3) = Omega_EDE0 * np.exp(-(z/zc)**m)
    # EDE é um campo escalar que decai rapidamente.
    Omega_de0_no_ede = 1 - Omega_m - f_EDE # Assumindo 1-Om é a Omega_DE total. f_EDE é uma fração que existe no início.
    # Essa parametrização de EDE é um pouco ambígua entre os seus arquivos.
    # Usaremos a que aparece em 'friedmann_model_fit.py.txt' para o termo Ω_EDE:
    # return Omega_EDE0 * np.exp(-(z/zc)**m)
    # Para o propósito de ter Omega_DE(0) = 1 - Omega_m, precisamos ajustar a Omega_EDE0
    # Ou Omega_EDE é uma *adição* e a Omega_Lambda é 1-Omega_m-Omega_EDE(0).
    # Vamos considerar que Omega_DE(z) = (1-Omega_m) + Omega_EDE_extra(z)
    # Ou que Omega_DE0 é a densidade de energia escura atual, e Omega_EDE modifica isso.
    # Considerando o `hybrid_cosmology.py` que tem `f_EDE` e `z_c` no `w_eff`.
    # É um campo escalar que tem w_eff dinâmico.
    # A implementação em `hybrid_cosmology.py` é mais completa para EDE.
    # Vamos usar a lógica de 'friedmann_model_fit.py.txt' que é mais um termo adicional
    # Omega_EDE(z, Omega_EDE0, zc, m=3)
    # Para o proposito de H(z), precisamos que Omega_DE(0) + Omega_m = 1
    # Se EDE é uma fração que domina no início, e depois decai.
    # Omega_DE(z) = (Omega_DE0 - f_EDE) * (Omega_de_lambda_part) + f_EDE * Omega_EDE_func(z)
    # Para simplicidade e consistência com 'friedmann_model_fit.py.txt':
    # O termo Omega_EDE é uma *adição*.
    # Omega_L_base = 1 - Omega_m
    # return Omega_L_base + f_EDE * np.exp(-(z/z_c)**3) # m=3 é um valor comum
    # Isso faria Omega_DE(0) = 1-Om + f_EDE, o que não é plano.
    # A forma como o 'hybrid_cosmology.py' lida com isso é mais sutil via w_eff.
    # Vamos usar uma versão simplificada de EDE, onde a Omega_DE(z) total tem um pico no passado.
    # Uma forma comum de EDE: Omega_DE(z) = Omega_DE0 * (1-f_EDE) + Omega_EDE_peak(z)
    # Onde Omega_EDE_peak(z) é um termo com um pico.
    # Para este script, vou usar uma simplificação que não é totalmente autoconsistente com a planicidade,
    # ou vou ter que assumir que Omega_m é parte de Omega_total e Omega_DE_0 é o resto.
    # Vou usar a forma que já está no 'friedmann_model_fit.py.txt' para Omega_EDE:
    return f_EDE * np.exp(-(z/z_c)**3) # Assume que este é um termo ADICIONAL.
                                        # Para manter a planicidade, a Omega_L base deve ser (1 - Omega_m - f_EDE * exp(-(0/z_c)**3))
                                        # Simplificando para que o termo principal seja (1 - Omega_m) e EDE seja um "extra"
                                        # Assim, Omega_DE(z) = (1-Omega_m) + Omega_EDE_term(z)
                                        # Isso vai exigir que Omega_m seja ajustado para manter a planicidade.
                                        # O `friedmann_model_fit.py.txt` ajusta H0, Omega_ond0, n, Omega_EDE0, zc.
                                        # E Omega_m e Omega_L são constantes (0.3 e 0.7).
                                        # Isso implica que a planicidade não é mantida automaticamente.
                                        # Para manter a planicidade (Ωm + ΩL + Ωk = 1), e se Ωk = 0, então ΩL = 1 - Ωm.
                                        # Aquele script tem: Omega_L = 0.7. Isso quer dizer que Ω_m e Ω_L são fixos e Ω_ond e Ω_EDE são adicionais.
                                        # Isso é uma violação da equação de Friedmann se Ω_k = 0.
                                        # Vou seguir a prática mais comum de que Ω_DE é o "resto" para a planicidade.
                                        # Ou seja, Ω_DE(z) = (1 - Ω_m - Ω_k) * f(z), onde f(z) é a evolução.

    # Para um modelo EDE que mantém a planicidade:
    # A densidade total de energia escura no presente é Omega_de0 = (1 - Omega_m).
    # A forma mais comum de EDE tem Omega_DE(z) = (1-Omega_m) * ( (1-f_EDE) + f_EDE * (1+z)^n_EDE * exp(...) )
    # Vamos seguir a lógica do hybrid_cosmology.py que tem um w_eff e uma Omega_DE(z) integrada.
    # Isso requer a classe HybridCosmology.
    # Por simplicidade, para este script, vou tratar EDE como um termo adicional que afeta a taxa de expansão,
    # e a Omega_Lambda implícita ajusta para a planicidade.
    # Esta é a parte mais tricky para compatibilizar.
    # Vou tentar uma abordagem simplificada, onde EDE é um "spike" de energia escura.
    Omega_de0_base = 1 - Omega_m # Parte da energia escura que se comporta como Lambda
    return Omega_de0_base + f_EDE * np.exp(-((1 + z) / z_c)**3) # Um termo que é significativo em z_c

# 2.4. Omega_ond Model (do tex.pdf e download.pdf)
def Omega_de_ond(z, Omega_m, Omega_ond0, n):
    Omega_de0_base = 1 - Omega_m # Parte da energia escura que se comporta como Lambda
    return Omega_de0_base + Omega_ond0 * (1 + z)**n

# 2.5. HIDEM (Halo-Induced Dark Energy Model)
def Omega_halo_HIDEM(z, Omega_seed, m, ac):
    a = 1 / (1 + z) # fator de escala
    return Omega_seed * (a**(-m)) * np.exp(-ac / a)

def Omega_de_HIDEM(z, Omega_m, Omega_seed, m, ac):
    # No HIDEM, Omega_halo(a) é o termo da energia escura.
    # A equação de Friedmann é H^2 = H0^2 [Omega_m a^-3 + Omega_halo(a)]
    # Implica que Omega_halo(a) é o Omega_DE(a).
    # Precisamos garantir a planicidade.
    # Se Omega_m é a matéria, e Omega_halo é a DE, então Omega_m + Omega_halo(0) = 1
    # Então, Omega_seed = 1 - Omega_m para z=0 (a=1).
    # Mas seus parâmetros ajustados para HIDEM são Omega_m=0.030, Omega_seed=0.903.
    # Isso já é 0.933, não 1.0. Isso pode significar que H0 não é o H0 "padrão",
    # ou que a densidade de radiação/curvatura não foi zero, ou que o modelo não é plano.
    # Ou que Omega_seed é a contribuição de DE em z=0 e é um parâmetro livre.
    # Vamos usar Omega_seed diretamente como um dos parâmetros do modelo de DE.
    # Então a H(z) é H0 * sqrt(Om * (1+z)^3 + Omega_halo_HIDEM(z, Omega_seed, m, ac))
    # E os parâmetros do fit seriam: H0, Omega_m, Omega_seed, m, ac.
    return Omega_halo_HIDEM(z, Omega_seed, m, ac)


# --- 3. Funções para Calcular Observáveis e Chi-Quadrado ---

# H(z) / H0 = E(z)
def E_model(z, H0, Omega_m, model_type, *model_params):
    if model_type == 'LCDM':
        # model_params deve ser vazio
        Omega_de_func = lambda z_val, om_val: Omega_de_LCDM(z_val, om_val)
        return np.sqrt(Omega_m * (1 + z)**3 + Omega_de_func(z, Omega_m))
    elif model_type == 'CPL':
        w0, wa = model_params
        Omega_de_func = lambda z_val, om_val: Omega_de_CPL(z_val, om_val, w0, wa)
        return np.sqrt(Omega_m * (1 + z)**3 + Omega_de_func(z, Omega_m))
    elif model_type == 'EDE':
        # f_EDE, z_c
        f_EDE, z_c = model_params
        # Aqui, a Omega_DE(z) precisa ser:
        # (1 - Omega_m - f_EDE) para garantir Omega_DE(0) = 1-Omega_m
        # e o termo EDE ser um spike.
        # OU, seguir o friedmann_model_fit.py.txt, onde EDE é ADICIONAL.
        # Se for adicional, a planicidade é questionável a menos que H0 seja ajustado.
        # Para compatibilidade com a maioria dos modelos, vou assumir (1-Omega_m) é o background.
        Omega_de0_base = 1 - Omega_m
        return np.sqrt(Omega_m * (1 + z)**3 + Omega_de0_base + f_EDE * np.exp(-((1 + z) / z_c)**3))
    elif model_type == 'Omega_ond':
        # Omega_ond0, n
        Omega_ond0, n = model_params
        Omega_de0_base = 1 - Omega_m
        return np.sqrt(Omega_m * (1 + z)**3 + Omega_de0_base + Omega_ond0 * (1 + z)**n)
    elif model_type == 'HIDEM':
        # Omega_seed, m, ac
        Omega_seed, m, ac = model_params
        # No HIDEM, Omega_m e Omega_halo são os componentes.
        # A soma no presente (z=0) deve ser 1.
        # Se Omega_m e Omega_seed são dadas como fixas no seu ajuste, então H0 é o parâmetro de escala.
        # No seu documento LaTeX do HIDEM, Omega_m e Omega_seed são 0.03 e 0.903, que somam 0.933.
        # Isso implica que 0.067 é curvatura, ou que H0 não é o H0 "real" que faz H^2(a)/H0^2 = Omega_total(a).
        # Vamos assumir que H0 é o parâmetro de escala, e (Omega_m + Omega_seed) é a densidade total efetiva no presente.
        # Então, E_model = sqrt(Omega_m*(1+z)^3 + Omega_halo_HIDEM(z, Omega_seed, m, ac))
        return np.sqrt(Omega_m * (1 + z)**3 + Omega_halo_HIDEM(z, Omega_seed, m, ac))
    else:
        raise ValueError(f"Tipo de modelo desconhecido: {model_type}")

# Distância de Luminosidade (para SNe Ia)
def dL(z, H0, Omega_m, model_type, *model_params):
    integrand = lambda z_: 1.0 / E_model(z_, H0, Omega_m, model_type, *model_params)
    integral, _ = quad(integrand, 0, z)
    d_c = (c / H0) * integral # Distância de co-movimento
    return (1 + z) * d_c # Distância de luminosidade

def mu_model(z_vals, H0, Omega_m, model_type, *model_params):
    return 5 * np.log10(dL(z_vals, H0, Omega_m, model_type, *model_params)) + 25

def chi2_sn(df_sn, H0, Omega_m, model_type, *model_params):
    mu_th = mu_model(df_sn['z'], H0, Omega_m, model_type, *model_params)
    return np.sum(((df_sn['mu'] - mu_th) / df_sn['sigma_mu'])**2)

# Distância de Diâmetro Angular (para BAO)
def dA(z, H0, Omega_m, model_type, *model_params):
    return dL(z, H0, Omega_m, model_type, *model_params) / (1 + z)**2

# Distância de Hubble (para BAO)
def dH(z, H0, Omega_m, model_type, *model_params):
    return c / (H0 * E_model(z, H0, Omega_m, model_type, *model_params))

# Distância de Dilatação de Volume (para BAO)
def dV(z, H0, Omega_m, model_type, *model_params):
    return ((c / H0 * quad(lambda x: 1 / E_model(x, H0, Omega_m, model_type, *model_params), 0, z)[0]) * dH(z, H0, Omega_m, model_type, *model_params)**2 * (1 + z)**2)**(1/3)


def chi2_bao(df_bao, H0, Omega_m, model_type, *model_params):
    # BAO é complexo. Geralmente envolve D_M/rd, D_H/rd, D_V/rd.
    # Assumindo que 'DM_over_rd' é o dado, precisamos calcular D_M e r_d para o modelo.
    # r_d é a escala do arrasto bariônico, depende de Omega_b, Omega_m, H0. É complexo de calcular precisamente.
    # Para simplificar, vamos assumir um r_d de referência ou fazer um ajuste no r_d.
    # No momento, como não temos a precisão do r_d, vou simplificar usando D_M/H0 (ou similar)
    # ou assumir que o "dados_bao" já é D_M(z)/r_d_fiducial.
    # Para o propósito deste script, vou fazer uma simplificação:
    # Apenas para que o chi2_bao funcione, vamos simular que df_bao['DM_over_rd'] corresponde a d_A(z_eff) / r_d_fiducial
    # Onde r_d_fiducial é uma constante.
    # Isso requer uma calibração ou um r_d fixo.
    # Para ser mais robusto, r_d precisa ser um parâmetro ou calculado a partir de Omega_b, etc.
    # Por ora, usaremos r_d_fiducial = 147.0 (valor comum).
    r_d_fiducial = 147.0 # Mpc
    DM_over_rd_th = dA(df_bao['z'], H0, Omega_m, model_type, *model_params) / r_d_fiducial
    return np.sum(((df_bao['DM_over_rd'] - DM_over_rd_th) / df_bao['sigma_DM_over_rd'])**2)

def chi2_hz(df_hz, H0, Omega_m, model_type, *model_params):
    Hz_th = H0 * E_model(df_hz['z'], H0, Omega_m, model_type, *model_params)
    return np.sum(((df_hz['Hz_obs'] - Hz_th) / df_hz['sigma_Hz'])**2)


# Função combinada de Chi-Quadrado total
def total_chi2(params, df_sn, df_bao, df_hz, model_type, param_names):
    # H0 é o primeiro parâmetro em todos os modelos.
    # Omega_m é o segundo parâmetro em todos os modelos (exceto HIDEM onde será o Omega_m do modelo).
    # O resto são os parâmetros específicos do modelo de DE.
    H0 = params[0]
    Omega_m_val = params[1] # Assumindo Omega_m é o 2o parâmetro.

    model_specific_params = params[2:]

    # Ajuste para o HIDEM: Omega_m é um dos 4 parâmetros ajustados.
    # No seu doc HIDEM: Omega_m=0.03, Omega_seed=0.903, m=0.993, ac=0.010 (4 params)
    # Mas H0 não está nessa lista.
    # Se H0 também é ajustado, são 5 parâmetros.
    # k = 4 no seu cálculo de AIC/BIC para HIDEM implica H0 não foi ajustado ou Omega_m não é um dos k.
    # Considerando o input do usuário (4 parâmetros), vou assumir que H0 é fixado para o HIDEM.
    # Para consistência, vou ajustar H0 para todos os modelos.
    # Então, os parâmetros de entrada serão: [H0, Omega_m, ..., model_params]

    chi2_val_sn = chi2_sn(df_sn, H0, Omega_m_val, model_type, *model_specific_params)
    chi2_val_bao = chi2_bao(df_bao, H0, Omega_m_val, model_type, *model_specific_params)
    chi2_val_hz = chi2_hz(df_hz, H0, Omega_m_val, model_type, *model_specific_params)

    # Aqui você pode adicionar o chi2 do CMB se tiver os dados e a função para ele
    # chi2_cmb = ...

    total_chi2_val = chi2_val_sn + chi2_val_bao + chi2_val_hz
    return total_chi2_val


# --- 4. Função para Calcular AIC e BIC ---
def compute_aic_bic(chi2, k, N):
    aic = chi2 + 2 * k
    bic = chi2 + k * np.log(N)
    return aic, bic

    bounds_ede = [(60, 80), (0.05, 0.5), (0.0, 0.2), (100, 5000)] # H0, Omega_m, f_EDE, z_c
    param_names_ede = ['H0', 'Omega_m', 'f_EDE', 'z_c']

    res_ede = minimize(total_chi2, initial_params_ede,
                       args=(df_sn, df_bao, df_hz, 'EDE', param_names_ede),
                       bounds=bounds_ede)

    chi2_ede = res_ede.fun
    k_ede = len(res_ede.x)
    aic_ede, bic_ede = compute_aic_bic(chi2_ede, k_ede, N_data_points)
    results['EDE'] = {'params': res_ede.x, 'chi2': chi2_ede, 'aic': aic_ede, 'bic': bic_ede, 'k': k_ede}
    print(f"Melhores parâmetros EDE: {res_ede.x}")
    print(f"Chi2 EDE: {chi2_ede:.2f}, AIC EDE: {aic_ede:.2f}, BIC EDE: {bic_ede:.2f}")
